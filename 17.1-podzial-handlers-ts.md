## Krok 0: `cover-cache.ts` — Współdzielony stan (nowy plik)

**Po co:** `coverResultCache`, `coverCacheLocks`, `durationCache`, `_store`, `PERSISTENT_COVER_DIR`, `COVER_CACHE_MAP_KEY` są używane przez `cover-handlers.ts`, `media-handlers.ts` i `library-handlers.ts`. Bez wydzielenia → circular imports lub duplikacja.

```typescript
// src/main/ipc/cover-cache.ts
import { stat, readFile, writeFile, mkdir, unlink } from 'fs/promises';
import { join } from 'path';
import { createHash } from 'crypto';
import os from 'os';
import { logger } from '../utils/logger';

let _store: InstanceType<typeof import('electron-store').default> | null = null;
export async function getStore() {
  if (!_store) {
    const { default: Store } = await import('electron-store');
    _store = new Store();
  }
  return _store;
}

export function getTempDir(): string {
  return join(os.tmpdir(), 'onda-covers');
}

export interface CachedCover {
  result: { type: 'video' | 'image' | null; data: string | null };
  mtimeMs: number;
}

export const coverResultCache = new Map<string, CachedCover>();
export const durationCache = new Map<string, { duration: number; mtimeMs: number }>();
export const coverCacheLocks = new Set<string>();

const CACHE_MAX_SIZE = 5000;

export function evictCache(map: Map<string, unknown>, maxSize: number): void {
  if (map.size <= maxSize) return;
  const toDelete = map.size - maxSize;
  let i = 0;
  for (const key of map.keys()) {
    if (i >= toDelete) break;
    map.delete(key);
    i++;
  }
}

export function cacheSet<T>(map: Map<string, T>, key: string, value: T, maxSize: number = CACHE_MAX_SIZE): void {
  map.set(key, value);
  evictCache(map as Map<string, unknown>, maxSize);
}

export const PERSISTENT_COVER_DIR = join(getTempDir(), 'persistent');
export const COVER_CACHE_MAP_KEY = 'coverCacheMap';

export function hashPath(filePath: string): string {
  return createHash('md5').update(filePath.toLowerCase()).digest('hex');
}

export async function getPersistentCover(
  filePath: string
): Promise<{ type: 'video' | 'image' | null; data: string | null } | null> {
  try {
    const store = await getStore();
    const cacheMap: Record<string, { cacheFile: string; mtime: number }> | undefined = store.get(COVER_CACHE_MAP_KEY) as any;
    const entry = cacheMap?.[filePath];
    if (!entry) return null;
    const s = await stat(filePath).catch(() => null);
    if (!s || s.mtimeMs > entry.mtime) {
      if (entry && cacheMap) { delete cacheMap[filePath]; store.set(COVER_CACHE_MAP_KEY, cacheMap); }
      return null;
    }
    const cachePath = join(PERSISTENT_COVER_DIR, entry.cacheFile);
    const buf = await readFile(cachePath).catch(() => null);
    if (!buf) return null;
    const isJpeg = entry.cacheFile.endsWith('.jpg');
    const dataUrl = isJpeg
      ? `data:image/jpeg;base64,${buf.toString('base64')}`
      : `data:image/png;base64,${buf.toString('base64')}`;
    return { type: 'image', data: dataUrl };
  } catch { return null; }
}

export async function savePersistentCover(filePath: string, binary: Buffer, ext: string): Promise<void> {
  try {
    await mkdir(PERSISTENT_COVER_DIR, { recursive: true });
    const hash = hashPath(filePath);
    const cacheFile = `${hash}.${ext}`;
    const cachePath = join(PERSISTENT_COVER_DIR, cacheFile);
    await writeFile(cachePath, binary);
    const s = await stat(filePath).catch(() => null);
    const store = await getStore();
    const cacheMap: Record<string, { cacheFile: string; mtime: number }> = (store.get(COVER_CACHE_MAP_KEY) as any) || {};
    cacheMap[filePath] = { cacheFile, mtime: s?.mtimeMs ?? Date.now() };
    store.set(COVER_CACHE_MAP_KEY, JSON.parse(JSON.stringify(cacheMap)));
  } catch { /* non-fatal */ }
}

export async function getCachedCover(
  filePath: string
): Promise<{ type: 'video' | 'image' | null; data: string | null } | null> {
  const memCached = coverResultCache.get(filePath);
  if (memCached) {
    try {
      const { mtimeMs } = await stat(filePath);
      if (mtimeMs <= memCached.mtimeMs) return memCached.result;
    } catch { /* file gone */ }
    coverResultCache.delete(filePath);
  }
  const diskCached = await getPersistentCover(filePath);
  if (diskCached) {
    const s = await stat(filePath).catch(() => null);
    cacheSet(coverResultCache, filePath, { result: diskCached, mtimeMs: s?.mtimeMs ?? Date.now() });
    return diskCached;
  }
  return null;
}
```

## Ryzyka i zabezpieczenia

### R1: `getMkvExtractPath` — cross-dependency
**Problem:** `subtitle-handlers.ts` używa `getMkvExtractPath()` zdefiniowanej w `dependency-handlers.ts`. Jeśli zrobisz circular import z `subtitle-handlers.ts` → `dependency-handlers.ts`, TypeScript zgłosi błąd.
**Rozwiązanie:** `getMkvExtractPath` jest eksportowana z `dependency-handlers.ts`, `subtitle-handlers.ts` importuje ją. Nie ma cyklu bo `dependency` nie importuje `subtitle`.

### R2: `electron-store` singleton (`_store`)
**Problem:** Lazy init `_store` w `cover-cache.ts` — pierwszy handler który go używa tworzy instancję. Jeśli `settings:get` zostanie wywołany przed `library:scan`, `_store` musi być ten sam.
**Rozwiązanie:** Singleton w `cover-cache.ts` — wszystkie pliki korzystają z `getStore()` z tego samego modułu.

### R3: `coverResultCache` / `durationCache` — współdzielony mutable stan
**Problem:** `cover-handlers.ts` zapisuje do `coverResultCache`, `media-handlers.ts` usuwa z niego przy `writeCover`, `library-handlers.ts` używa `durationCache`.
**Rozwiązanie:** Wszystkie operacje na cache przez `cover-cache.ts` — eksportowane `Map` + `cacheSet()`.

### R4: `uniqueId()` — duplikacja
**Problem:** `uniqueId()` potrzebne w `cover-handlers.ts` i `subtitle-handlers.ts`.
**Rozwiązanie:** Przenieść do `cover-cache.ts` lub zdefiniować lokalnie w każdym pliku (4 linie, niskie ryzyko).

### R5: `getTempDir()` — duplikacja
**Problem:** Używane w `cover-handlers.ts` + `subtitle-handlers.ts` + oryginalnie w `handlers.ts`.
**Rozwiązanie:** Już w `cover-cache.ts` — importować przez `cover-handlers.ts` i `subtitle-handlers.ts`.

### R6: `errMsg()` — import
**Problem:** Używany w `media-handlers.ts` (writeTags, writeCover, renameFile).
**Rozwiązanie:** Import z `../../shared/helpers` — już istnieje.

### R7: Regresja typu `dialog:saveFile` (return type)
**Problem:** W `index.d.ts` i `shared/types/ipc.ts` typ zwracany musi być zgodny z `Electron.SaveDialogReturnValue`.
**Rozwiązanie:** Przenieść `dialog:saveFile` z `fs-handlers.ts` do `dialog-handlers.ts` — jeden plik, jeden typ.

### R8: `SharpService` — ścieżka względna
**Problem:** `../utils/sharp` zmienia się gdy plik jest w podfolderze ipc/ → `../utils/sharp` jest poprawne zarówno z `handlers.ts` (root ipc/) jak i z nowych plików (też w ipc/).
**Rozwiązanie:** Wszystkie pliki w `src/main/ipc/` — ścieżka `../utils/sharp` działa dla wszystkich.
